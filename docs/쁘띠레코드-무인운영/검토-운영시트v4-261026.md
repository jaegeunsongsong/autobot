### 수정 1 — 행마다 try로 감싸고 날짜·시간 형식 검사
```js
function toDate(d, t) {
  let day;
  if (d instanceof Date) day = new Date(d.getFullYear(), d.getMonth(), d.getDate());
  else {
    const p = String(d).trim().replace(/[./]/g, '-').split('-').map(Number);
    if (p.length < 3) throw new Error('이용일은 2026-10-26 형식으로');
    day = new Date(p[0], p[1] - 1, p[2]);
  }
  let h = 0, m = 0;
  if (t instanceof Date) { h = t.getHours(); m = t.getMinutes(); }
  else if (t) { const p = String(t).trim().split(':'); h = Number(p[0]); m = Number(p[1] || 0); }
  if (isNaN(h) || isNaN(m)) throw new Error('시간은 14:00 형식으로');
  day.setHours(h, m, 0, 0);
  if (isNaN(day.getTime())) throw new Error('이용일·시간 확인');
  return day;
}
```

function checkDelivered(sh, rowNo, col, cellText) {

  const res = UrlFetchApp.fetch('https://api.solapi.com/messages/v4/groups/' + m[1],
    { headers: { Authorization: authHeader() }, muteHttpExceptions: true });
  if (res.getResponseCode() >= 300) return;                // 조회 실패는 그냥 둠
  const c = (JSON.parse(res.getContentText()).count) || {};
  if (c.sentFailed > 0 && !c.sentSuccess) sh.getRange(rowNo, col).setValue('실패: 미전달(알림톡·문자 모두) ' + m[1]);
  else if (c.sentSuccess > 0) sh.getRange(rowNo, col).setValue(String(cellText).replace(/ #\S+$/, ' ✓'));
}

### 수정 9 — 보안 운영 규칙(코드 아님)
- 시트는 대표·남편 두 계정에만 '편집자'로. "링크가 있는 모든 사용자"는 절대 금지. 편집자는 Apps Script를 열어 속성(API 키)을 볼 수 있음.
- 두 계정 모두 2단계 인증. 키가 샌 의심이 들면 솔라피에서 Secret 재생성(가이드 7절 그대로).
- 시트에 비밀번호·전화번호가 있으므로 '이용완료' 30일 뒤 비밀번호 칸은 지우는 월 1회 정리 추가.
- 더 막으려면: 스크립트를 시트에 묶지 말고 **독립 스크립트**로 만들어 `SpreadsheetApp.openById('시트ID')`로 열기 → 시트 편집자가 코드·속성을 못 봄. 대신 메뉴 `onOpen`이 없어지므로 실행은 트리거와 스크립트 편집기에서.

## 1인 운영자가 실수하기 쉬운 지점

| 지점 | 무슨 일이 생기나 | 막는 법 |
|---|---|---|
| 상태를 '확정'으로 바꾸고 비밀번호를 안 적음 | 지금은 "실패"로 굳음 | 수정 2 |
| 종료 시각을 안 적음 | 종료 00:00 → ②가 이용일 자정 기준으로 이미 지난 것으로 보고 **① 직후 다음 실행에 바로 나감** | 종료 빈칸이면 ① 발송 때 '실패: 종료 시간 없음' (수정 1의 검사에 `if (!t) throw`) |
| 시트 상태를 먼저 '이용완료'로 바꿈 | ② 안 나감 | 수정 3 |
| 테스트 줄을 안 지우고 TEST_MODE=N | 테스트 줄의 ②가 손님 번호가 아닌 TEST_PHONE으로 이미 나갔으니 문제 없음. 단 실제 손님 줄을 TEST_MODE=Y 상태에서 '확정'하면 손님은 못 받고 결과는 "성공(테스트)" | TEST_MODE=Y일 때 시트 제목 행을 주황으로 바꾸는 한 줄: `sh.getRange(1,1,1,HEAD.length).setBackground(test ? '#e69138' : '#6D2E46')` 를 run 시작에 |
| 결과 칸을 지워 재발송 | ①은 비밀번호 문자가 한 번 더 가는 정도. ②는 수정 3 전엔 안 나감 | 수정 3·4 |
| 10분 트리거가 꺼짐(권한 만료·계정 변경) | 아무것도 안 나가는데 시트는 조용함 | 매일 아침 "마지막 실행" 시각을 시트 Z1에 적는 한 줄: `sh.getRange(1, HEAD.length).setNote('마지막 실행 ' + fmtStamp(now))` + 전날 21시 루틴에 확인 |
| 대체 문자가 발신번호 9870으로 나감 | 손님 답장이 9870 폰으로 옴(가이드 7절) | 9870 폰 알림을 대표가 보는지 확인 |
